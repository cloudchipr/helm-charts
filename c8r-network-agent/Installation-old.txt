# Installation

1. Apply conntrack-enabler.yaml for your cloud(AWS or Azure)
2. Create namespaces
    kubectl create namespace kubenetmon-server
    kubectl create namespace kubenetmon-agent

3. Deploy kube-net-mon server
    helm install kubenetmon-server kubenetmon/kubenetmon-server \
    --namespace kubenetmon-server \
    --set region=eastus \
    --set cluster=my-aks-cluster \
    --set environment=production \
    --set cloud=azure \
    --set inserter.endpoint=ch-k8s-staging.cloudchipr.dev:9000 \
    --set inserter.username=default \
    --set inserter.password='YOUR_PASSWORD' \
    --set inserter.disableTLS=true

4. Deploy kube-net-mon agent
    helm install kubenetmon-agent kubenetmon/kubenetmon-agent \
  --namespace kubenetmon-agent
